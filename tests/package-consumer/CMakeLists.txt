cmake_minimum_required(VERSION 3.21)
project(QuadTreeConsumer LANGUAGES CXX)

if (QUADTREE_TEST_MODE STREQUAL "subdirectory")
    add_subdirectory("${QUADTREE_TEST_SOURCE}" quadtree)
elseif (QUADTREE_TEST_MODE STREQUAL "fetchcontent")
    include(FetchContent)
    FetchContent_Declare(quadtree SOURCE_DIR "${QUADTREE_TEST_SOURCE}")
    FetchContent_MakeAvailable(quadtree)
elseif (QUADTREE_TEST_MODE STREQUAL "installed")
    find_package(QuadTree 2.0 CONFIG REQUIRED)
else()
    message(FATAL_ERROR "Set QUADTREE_TEST_MODE to subdirectory, fetchcontent, or installed")
endif()

get_target_property(features QuadTree::QuadTree INTERFACE_COMPILE_FEATURES)
if (NOT features STREQUAL "cxx_std_20")
    message(FATAL_ERROR "Unexpected compile features: ${features}")
endif()
foreach(property INTERFACE_COMPILE_OPTIONS INTERFACE_COMPILE_DEFINITIONS INTERFACE_LINK_LIBRARIES INTERFACE_LINK_OPTIONS)
    get_property(value TARGET QuadTree::QuadTree PROPERTY ${property})
    if (NOT "${value}" STREQUAL "")
        message(FATAL_ERROR "Unexpected ${property}: ${value}")
    endif()
endforeach()

if (QUADTREE_TEST_MODE STREQUAL "installed")
    cmake_path(IS_PREFIX QUADTREE_TEST_PREFIX "${QuadTree_DIR}" NORMALIZE config_in_prefix)
    get_target_property(imported QuadTree::QuadTree IMPORTED)
    get_target_property(include_dirs QuadTree::QuadTree INTERFACE_INCLUDE_DIRECTORIES)
    if (NOT imported OR NOT config_in_prefix OR NOT include_dirs)
        message(FATAL_ERROR "Expected an imported package under ${QUADTREE_TEST_PREFIX}")
    endif()
    foreach(dir IN LISTS include_dirs)
        cmake_path(IS_PREFIX QUADTREE_TEST_PREFIX "${dir}" NORMALIZE include_in_prefix)
        if (NOT include_in_prefix)
            message(FATAL_ERROR "Include path escapes the installed prefix: ${dir}")
        endif()
    endforeach()
else()
    if (QUADTREE_BUILD_DEMO OR QUADTREE_BUILD_TESTS OR QUADTREE_BUILD_BENCHMARKS OR QUADTREE_INSTALL)
        message(FATAL_ERROR "QuadTree subproject enabled an optional build or installation by default")
    endif()
endif()

add_executable(consumer main.cpp)
set_target_properties(consumer PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(consumer PRIVATE QuadTree::QuadTree)
enable_testing()
add_test(NAME quadtree-consumer COMMAND consumer)
